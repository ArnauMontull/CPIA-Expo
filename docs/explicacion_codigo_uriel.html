4. Aplicación de DBSCAN Paso a Paso (Líneas 46 – 89)

EPS = 0.65
MIN_SAMPLES = 3

def ejecutar_dbscan(X, eps=0.65, min_samples=3):
    try:
        from sklearn.cluster import DBSCAN as SklearnDBSCAN
        return SklearnDBSCAN(eps=eps, min_samples=min_samples).fit_predict(X)
    except Exception:
        N = len(X)
        etiquetas = np.full(N, -1, dtype=int)
        visitados = np.zeros(N, dtype=bool)
        cluster_actual = 0

        D = np.linalg.norm(X[:, None, :] - X[None, :, :], axis=-1)

        for i in range(N):
            if visitados[i]: continue
            visitados[i] = True

            vecinos = np.where(D[i] <= eps)[0]
            if len(vecinos) < min_samples:
                etiquetas[i] = -1
            else:
                etiquetas[i] = cluster_actual
                cola = list(vecinos[vecinos != i])
                while cola:
                    j = cola.pop(0)
                    if not visitados[j]:
                        visitados[j] = True
                        j_vecinos = np.where(D[j] <= eps)[0]
                        if len(j_vecinos) >= min_samples:
                            for v in j_vecinos:
                                if v not in cola and etiquetas[v] == -1:
                                    cola.append(v)
                    if etiquetas[j] == -1:
                        etiquetas[j] = cluster_actual
                cluster_actual += 1
        return etiquetas

df["cluster"] = ejecutar_dbscan(X_scaled, eps=EPS, min_samples=MIN_SAMPLES)

5. Evaluación con Silhouette Score (Líneas 94 – 120)

def calcular_silhouette(X, labels):
    mask = labels != -1
    if len(set(labels[mask])) <= 1: return None
    D = np.linalg.norm(X[:, None, :] - X[None, :, :], axis=-1)
    sils = []
    for i in range(len(X)):
        if labels[i] == -1: continue
        c_i = labels[i]
        same_c = np.where((labels == c_i) & (np.arange(len(X)) != i))[0]
        a_i = np.mean(D[i, same_c]) if len(same_c) > 0 else 0
        other_clusters = set(labels) - {-1, c_i}
        b_i = min(np.mean(D[i, labels == c_o]) for c_o in other_clusters)
        sils.append((b_i - a_i) / max(a_i, b_i))
    return float(np.mean(sils))
  • Línea 95: mask = labels != -1 excluye los puntos de ruido para no falsear la silueta interna de los clusters densos.
  • Línea 106: Para cada material i calcula a_i (distancia media a los materiales de su mismo cluster → cohesión).
  • Línea 108: Calcula b_i (distancia media al cluster vecino más cercano → separación).
  • Línea 109: Aplica la fórmula exacta s = (b - a) / max(a, b). El promedio resultante da 0.7858, certificando una estructura de clusters excelente.
